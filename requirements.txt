# Client quick-install for ttsgen / ttsplay / ttsrec / ttsapi.
# Mirrors pyproject.toml [project].dependencies — keep in sync.
#
# Heavier deps live in optional extras (NOT installed by `pip install -r`):
#   pip install ".[piper]"    → piper-tts                                   (pipertts engine)
#   pip install ".[silero]"   → torch + torchaudio + omegaconf              (silerotts engine)
#   pip install ".[coqui]"    → coqui-tts[codec] + torch + transformers     (coquitts engine; Python 3.9–3.12)
#   pip install ".[bark]"     → bark (git) + scipy + numpy                  (barktts engine; ~10–15 GB models)
#   pip install ".[api]"      → Flask + flask-cors + marshmallow + gunicorn + uvicorn + asgiref + pytz
#                              (Flask deps for running ttssrv/app1.py outside Docker)
#   pip install ".[dev]"      → black + flake8 + pytest + pytest-cov          (development tools)
#   pip install ".[all]"      → piper + silero + api                         (combined common set)
#
# `ttsgen --install <engine>` is the recommended path — it picks the right
# torch index (cpu / cu121) and persists the model directory to ~/.config/ttsgen.conf.
#
# Server (ttssrv) is normally deployed via Docker:
#   docker compose up                                    # GPU image
#   docker compose -f docker-compose-cpu.yml up          # CPU-only image
